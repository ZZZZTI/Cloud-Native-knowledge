### 内置语法

```Shell
# 接受用户输入
read [选项] 变量
read -s               # 静默输入
read -p "请输入"       # 带提示输入
read -t 5             # 输入限时
read -n 1             # 输入限字数

# 输出
echo -n "..."         # 不换行输出
echo "..."            # 输出并解析变量
echo '...'            # 原样输出字符串
echo "error" >&2      # 报错信息扔掉
printf "时间:%d\n" 25  # %d->25
cat/命令 << EOF
...............
...............
...............
EOF                   # 输出长文本
seq -F'正则'/-s'分隔符'  [起始数 +-步长 末尾数]  # 打印数字



# 日期
date                  # 显示当前时间
date -r <文件名>       # 文件最后修改时间 
date -d "2023-04-11 + 2 weeks 3 days" # 计算时间
date +"%Y-%m-%d %H:%M:%S"             # 自定义时间格式

# 随机数
$RANDOM               # 0-32767
$((RANDOM % 101))     # 0-100
$((RANDOM % 51 + 50)) # 50-100

exit (0/1)            # 退出脚本
```


## 一、字符串处理

### 长度、截取、拼接

```bash
s="hello world"

echo ${#s}              # 长度 → 11
echo ${s:0:5}           # 从 0 取 5 个 → hello
echo ${s:6}             # 从 6 到末尾 → world
echo ${s:6:3}           # 从 6 取 3 个 → wor
echo ${s: -5}           # 取最后 5 个 → world（负数前必须有空格）

# 拼接
a="hello"; b="world"
echo "${a} ${b}"        # hello world
echo "${a}${b}"         # helloworld
name="web"; echo "${name}-01"    # 变量名后接字符要用 ${}
```

### 删除前缀与后缀（最常用）

```bash
f="/var/log/nginx/access.log"

echo ${f#*/}            # var/log/nginx/access.log    删「最短」前缀匹配
echo ${f##*/}           # access.log                  删「最长」前缀 → basename
echo ${f%/*}            # /var/log/nginx              删「最短」后缀 → dirname
echo ${f%%/*}           # （空）                       删「最长」后缀

echo ${f%.*}            # /var/log/nginx/access       去掉最后一个扩展名
echo ${f%%.*}           # /var/log/nginx/access       去掉所有扩展名
echo ${f##*.}           # log                         取扩展名

# 记法：# 删左边（前缀），% 删右边（后缀）；单个是「最短匹配」，两个是「最长匹配」
```

### 替换

```bash
s="hello world"

echo ${s/world/linux}   # hello linux     替换第一个匹配
echo ${s//o/0}          # hell0 w0rld     替换全部
echo ${s/#hello/hi}     # hi world        只替换开头
echo ${s/%world/linux}  # hello linux     只替换结尾

# 大小写（bash 4.0+）
echo ${s^^}             # HELLO WORLD
echo ${s,,}             # hello world
echo ${s^}              # Hello world     首字母大写
```

### 默认值与判空

```bash
echo ${x:-default}      # x 未设置或为空 → 输出 default（不赋值）
echo ${x:=default}      # 同上，并且把 default 赋给 x
echo ${x:+set}          # x 已设置且非空 → 输出 set，否则输出空
echo ${x:?must set}     # x 为空 → 输出错误信息并退出脚本（退出码 1）

# 典型用法：必填参数校验
: ${1:?用法: $0 <namespace>}
```

### 实用组合

```bash
# 去掉路径取文件名（不依赖 basename）
name="${path##*/}"

# 取不带扩展名的文件名
base="${name%.*}"

# 去掉 Windows 换行符
line="${line%$'\r'}"

# 判断是否为整数
[[ "$n" =~ ^[0-9]+$ ]] && echo "是整数"

# 判断是否为 IP
[[ "$ip" =~ ^([0-9]{1,3}\.){3}[0-9]{1,3}$ ]] && echo "形如 IP"

# 去首尾空格
s="  abc  "
echo "[${s#"${s%%[![:space:]]*}"}]"          # 去左侧
echo "[${s%"${s##*[![:space:]]}"}]"          # 去右侧
```

## 二、数组

> ⚠️ 版本要求：关联数组、`mapfile`、`${s^^}` 大小写转换都需要 **bash 4.0+**。
> RHEL/CentOS 8+ 自带 4.4/5.1 没问题；**macOS 自带的是 bash 3.2**，本地测试会报
> `declare: -A: invalid option`、`mapfile: command not found`，需先 `brew install bash`。
> 用 `bash --version` 确认。

### 普通数组

```bash
# 定义
arr=(a b c)
arr[3]=d
arr2=("a b" c)          # 元素含空格必须加引号
declare -a arr3=(x y)   # 显式声明

# 取值
echo ${arr[0]}          # a
echo ${arr[@]}          # 全部元素
echo ${#arr[@]}         # 元素个数 → 4
echo ${!arr[@]}         # 所有下标 → 0 1 2 3
echo ${arr[@]:1:2}      # 切片：从下标 1 取 2 个
echo ${arr[-1]}         # 最后一个（bash 4.3+）

# 追加与删除
arr+=(e f)              # 追加多个元素
unset 'arr[1]'          # 删除下标 1（引号必须加）
```

### 遍历

```bash
for x in "${arr[@]}"; do
    echo "$x"
done

# 带下标遍历
for i in "${!arr[@]}"; do
    echo "$i = ${arr[$i]}"
done
```

### 关联数组（bash 4+）

```bash
declare -A map
map[host]=10.0.0.1
map[port]=22
map[user]=root

echo ${map[host]}       # 10.0.0.1
echo ${#map[@]}         # 键的数量
echo ${!map[@]}         # 所有键
unset 'map[user]'

for k in "${!map[@]}"; do
    echo "$k = ${map[$k]}"
done
```

### 常用场景

```bash
# 1. 批量操作主机
hosts=(10.0.0.1 10.0.0.2 10.0.0.3)
for h in "${hosts[@]}"; do
    ping -c1 -W1 "$h" >/dev/null && echo "$h up" || echo "$h down"
done

# 2. 并发执行
for h in "${hosts[@]}"; do
    ping -c1 -W1 "$h" >/dev/null &
done
wait

# 3. 把文件读进数组（推荐 mapfile）
mapfile -t lines < /etc/hosts
echo "${#lines[@]} 行"
echo "${lines[0]}"

# 4. 把命令输出读进数组
mapfile -t pods < <(kubectl get pod -o name)

# 5. 数组转字符串（指定分隔符）
IFS=','; echo "${arr[*]}"; unset IFS
```

### 坑

```bash
"${arr[@]}"     # ✅ 每个元素单独一个词，保留元素内的空格
${arr[@]}       # ❌ 会被再次按空格切分
${arr[*]}       # 用 IFS 首字符连接成一个词

arr=(1 2 3)
echo ${#arr}    # 3（等价于元素个数，但对字符串是长度，易混）
```

## 三、数值与日期

### 整数运算

```bash
echo $(( 7 / 2 ))       # 3     整数除法，直接截断
echo $(( 7 % 2 ))       # 1
echo $(( 2 ** 10 ))     # 1024
echo $(( (1 + 2) * 3 ))

i=1
((i++));  echo $i       # 2
((i+=5)); echo $i       # 7
let "i*=2"; echo $i     # 14

# 十六进制
echo $(( 16#ff ))       # 255
printf '%d\n' 0xff      # 255
```

### 浮点运算（bash 不支持，用 bc 或 awk）

```bash
# bc：scale 控制小数位
echo "scale=2; 10/3" | bc               # 3.33
echo "scale=4; 1/7"  | bc               # .1428
awk 'BEGIN{printf "%.2f\n", 10/3}'      # 3.33
awk 'BEGIN{print 10/3}'                 # 3.33333

# 比较浮点
if awk "BEGIN{exit !(1.5 > 1.2)}"; then
    echo "1.5 更大"
fi

# 百分比
used=87; total=100
awk -v u=$used -v t=$total 'BEGIN{printf "%.1f%%\n", u/t*100}'
```

### 日期

```bash
date                            # 默认格式
date '+%F %T'                   # 2026-11-23 10:30:00
date '+%Y%m%d_%H%M%S'           # 20261123_103000（做文件名）
date +%s                        # 时间戳
date -d @1700000000 '+%F %T'    # 时间戳转日期

# 日期加减（GNU date）
date -d '7 days ago' '+%F'
date -d 'yesterday' '+%F'
date -d 'tomorrow' '+%F'
date -d '2026-01-01 +30 days' '+%F'
date -d 'next monday' '+%F'

# macOS 是 BSD date，写法不同
date -v-7d '+%F'
date -v+1m '+%F'

# 算天数差
d1=$(date -d '2026-01-01' +%s)
d2=$(date +%s)
echo $(( (d2 - d1) / 86400 ))

# 计时
start=$(date +%s%N)
sleep 1
echo "$(( ($(date +%s%N) - start) / 1000000 )) ms"

SECONDS=0
sleep 2
echo "${SECONDS}s"
```
