### 脚本头部模版

```Shell
#!/usr/bin/env bash     
export PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin
set -euo pipefail
# =================================
# 脚本名称: script.sh
# 功能描述: .........
# 作者   : nbbro
# 创建日期: 2026-06-18
# 版本   : 1.0
# =================================
# ---------- 全局变量配置 -----------
readonly BACKUP_SOURCE="/data"
readonly BACKUP_DEST="/backup"
readonly LOG_FILE="/var/log/backup.log"
readonly NAME_NUM=10
# ---------- 引入其他库 -------------
source /path/to/lib1.sh
source /path/to/lib2.sh
# ---------- 函数定义 ---------------
function func1() {...}
function func2() {...}
# ---------- 主逻辑 ----------------
main(){
    ....
}

main "$@"
```

### 执行脚本

```Shell
bash script.sh a b c     # a,b,c为用户传入的位置参数，供脚本内部调用
bash -n script.sh        # 检查语法,不实际运行
bash -x script.sh        # 打印每一行执行的命令及其参数
source ./script.sh
```




## 八、调试与规范

### 语法检查与跟踪

```bash
bash -n script.sh           # 只检查语法，不执行
bash -x script.sh           # 打印每条执行的命令（前面带 +）
bash -v script.sh           # 打印读入的每一行
bash -u script.sh           # 使用未定义变量时报错

# 只跟踪某一段
set -x
...要调试的代码...
set +x
```

### 严格模式（脚本开头必备）

```bash
#!/usr/bin/env bash
set -euo pipefail
IFS=$'\n\t'

# -e       命令失败立即退出
# -u       使用未定义变量报错
# -o pipefail  管道中任一环节失败即整体失败
# IFS      防止用未加引号的变量做单词分割
```

```bash
# -e 的例外：某些命令允许失败
command || true
if ! command; then :; fi
```

### trap 清理

```bash
TMP=$(mktemp -d)
trap 'rm -rf "$TMP"' EXIT           # 退出时清理，无论成功失败

# 多点清理 + 信号处理
cleanup() {
    echo "清理中..." >&2
    rm -f "$LOCK"
}
trap cleanup EXIT INT TERM

# 加锁防并发（配合 flock）
exec 9>"$LOCK"
flock -n 9 || { echo "已有实例在跑"; exit 1; }
```

### 日志与输出规范

```bash
# 日志函数
log() {
    printf '%s [%s] %s\n' "$(date '+%F %T')" "$1" "$2" >&2
}
log INFO  "开始巡检"
log ERROR "连接失败: $host"

# 输出约定
# 正常结果 → stdout
# 日志/错误 → stderr（重定向到文件时不会污染结果）
# 退出码   0 成功 / 1 检查不通过 / 2 执行失败
```

### 参数与校验

```bash
# 位置参数校验
if [ $# -lt 1 ]; then
    echo "用法: $0 <namespace>" >&2
    exit 2
fi

# 或者用默认值语法一行搞定
: "${1:?用法: $0 <namespace>}"

# 文件必须存在
[[ -f "$conf" ]] || { echo "配置不存在: $conf" >&2; exit 2; }

# 命令必须存在
command -v kubectl >/dev/null || { echo "缺少 kubectl" >&2; exit 2; }
```

### shellcheck

```bash
# 安装
brew install shellcheck     # macOS
dnf install ShellCheck      # RHEL

# 使用
shellcheck script.sh
shellcheck -S warning script.sh     # 只报 warning 以上
shellcheck -f gcc script.sh         # 便于编辑器跳转

# 常见告警
# SC2086  变量未加引号 → "$var"
# SC2181  不要用 $? 判断，直接 if command; then
# SC2002  cat file | grep → 直接 grep file
# SC2164  cd 失败要继续 → cd dir || exit
```

```Shell
mktemp [选项] [mytemp/dir.XXXXXX]
mktemp -t	           # 在/tmp下创建
mktemp -d	           # 创建临时目录
mktemp -p <dir>	     # 指定创建位置	 

trap '清理命令1;命令2;函数' [信号列表]
```
