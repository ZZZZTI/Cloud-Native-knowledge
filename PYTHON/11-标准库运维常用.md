## 1. os 与 sys：和操作系统对话

```python
import os, sys

# 环境变量（运维脚本的主要配置来源）
os.environ.get("HOME")                    # 取不到返回 None
os.environ.get("PORT", "8080")            # 给默认值
os.environ["API_TOKEN"]                   # 取不到直接 KeyError（必需项可用它）
os.environ["DEBUG"] = "1"                 # 只影响当前进程

# 路径（能用 pathlib 就用 pathlib，见 08 章）
os.path.join("/var", "log", "a.log")
os.path.basename("/var/log/a.log")        # 'a.log'
os.path.dirname("/var/log/a.log")         # '/var/log'
os.path.abspath("./a.txt")                # 绝对路径

# 目录操作
os.getcwd()                               # 当前目录
os.chdir("/tmp")                          # 切换（少用，容易出 bug）
os.makedirs("/tmp/a/b", exist_ok=True)
os.listdir("/var/log")
os.walk("/var/log")                       # 递归遍历：返回 (当前目录, 子目录列表, 文件列表)

# 进程信息
os.getpid()                               # 自己的 PID
os.getppid()
os.cpu_count()                            # CPU 核数
os.getloadavg()                           # 负载（1/5/15 分钟）

# 执行系统命令的轻量方式（不需要输出时）
os.system("date")                         # 返回退出码，输出直接打屏（不推荐）
```

```python
import sys

sys.argv                # 命令行参数列表，argv[0] 是脚本名
sys.version             # Python 版本字符串
sys.platform            # 'darwin' / 'linux'
sys.exit(0)             # 退出脚本，0 表示成功
sys.stdout.write("...") # 不换行输出
sys.stderr.write("...") # 输出到标准错误（日志/告警应走这里）
```

## 2. subprocess：调用外部命令（重点）

```python
import subprocess

# ✅ 最常用：跑完拿结果，失败抛异常
result = subprocess.run(
    ["kubectl", "get", "pods", "-o", "json"],   # 列表传参，不要拼字符串
    capture_output=True,                        # 捕获 stdout/stderr
    text=True,                                  # 返回字符串而不是 bytes
    timeout=30,                                 # 必须有超时
    check=True,                                 # 非 0 退出码抛 CalledProcessError
)
print(result.stdout)
```

```python
# 不想抛异常：自己看 returncode
r = subprocess.run(["systemctl", "is-active", "nginx"],
                   capture_output=True, text=True, timeout=5)
if r.returncode != 0:
    print("nginx 未运行:", r.stderr.strip())
```

```python
# 需要管道/重定向时才用 shell=True
subprocess.run("ps aux | grep nginx | wc -l", shell=True, text=True,
               capture_output=True, timeout=10)
# ⚠️ 用了 shell=True 就绝不能把外部输入拼进命令（命令注入）
```

```python
# 逐行读输出（适合长时间运行的命令，如 tail -f）
p = subprocess.Popen(["journalctl", "-u", "kubelet", "-f"],
                     stdout=subprocess.PIPE, text=True, bufsize=1)
try:
    for line in p.stdout:
        print(line.rstrip())
finally:
    p.terminate()          # 记得清理子进程
    p.wait(timeout=5)
```

**四条纪律**：

1. **用列表传参**，不用字符串拼接（避免空格和注入问题）。
2. **必设 `timeout`**，否则命令挂住脚本就挂住。
3. **`text=True`**，省去手动 decode。
4. **要判断结果就用 `check=True` 或自己看 `returncode`**，不要假设一定成功。

## 3. shutil：文件与磁盘

```python
import shutil

shutil.copy2("a.conf", "b.conf")        # 复制（保留时间戳等元数据）
shutil.copytree("/src", "/dst", dirs_exist_ok=True)
shutil.move("/tmp/x", "/data/x")
shutil.rmtree("/tmp/cache", ignore_errors=True)
shutil.chown("/data/f", user="app", group="app")

# 磁盘水位（巡检高频）
total, used, free = shutil.disk_usage("/data")
print(f"使用率 {used / total:.1%}，剩余 {free / 1024**3:.1f} GiB")
```

## 4. argparse：命令行参数（脚本标配）

```python
import argparse

p = argparse.ArgumentParser(
    description="检查集群中异常的 Pod",
    formatter_class=argparse.RawDescriptionHelpFormatter,
)
p.add_argument("namespace", help="命名空间")                       # 位置参数（必填）
p.add_argument("-l", "--label", default="app=web", help="标签选择器")
p.add_argument("-n", "--retries", type=int, default=3, help="重试次数")
p.add_argument("--dry-run", action="store_true", help="只打印不执行")   # 布尔开关
p.add_argument("-v", "--verbose", action="count", default=0, help="可叠加 -vv")

args = p.parse_args()

print(args.namespace)      # 字符串
print(args.retries)        # int（type=int 自动转换）
print(args.dry_run)        # True / False
```

```bash
python3 check.py prod -l app=db -n 5 --dry-run -vv
python3 check.py --help        # 自动生成帮助，免费
```

**为什么不用 `input()`**：脚本通常无人值守（cron / K8s Job），交互式输入会挂住。参数化才能被调度系统调用。

## 5. logging：比 print 强在哪

```python
import logging

logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s %(levelname)-5s %(name)s %(message)s",
    datefmt="%Y-%m-%d %H:%M:%S",
)
log = logging.getLogger("inspect")     # 用模块名当 logger 名，便于定位来源

log.debug("默认不输出")
log.info("开始检查 %s", "prod")
log.warning("磁盘使用率 %d%%", 85)
log.error("连接失败: %s", err)
```

```python
# 同时输出到文件 + 自动轮转（长期运行必加）
from logging.handlers import RotatingFileHandler

fh = RotatingFileHandler("/var/log/inspect.log",
                         maxBytes=10 * 1024 * 1024,   # 10MB 切一个
                         backupCount=5)               # 保留 5 个
fh.setFormatter(logging.Formatter("%(asctime)s %(levelname)s %(message)s"))
log.addHandler(fh)
```

**用 `%s` 占位而不是 f-string**：只有真正输出时才格式化，DEBUG 级别的日志在不输出时零开销。

```python
log.debug("大对象: %s", huge_obj)      # ✅ 惰性求值
log.debug(f"大对象: {huge_obj}")       # ❌ 无论输不输出都会先做字符串拼接
```

## 6. datetime：时间是最大的坑

```python
from datetime import datetime, timedelta, timezone

# 带时区（推荐）
now = datetime.now(timezone.utc)
local = datetime.now().astimezone()

# 格式化与解析
now.strftime("%Y-%m-%d %H:%M:%S")
datetime.strptime("2026-10-27 10:00:00", "%Y-%m-%d %H:%M:%S")

# 时间运算
now - timedelta(hours=1)
now + timedelta(days=7)

# 时间差
delta = datetime(2026, 11, 1) - datetime(2026, 10, 27)
delta.days          # 5
delta.total_seconds()

# 时间戳互转
now.timestamp()                              # → 浮点秒
datetime.fromtimestamp(1730000000, timezone.utc)
```

**统一规则**：**内部一律用 UTC 存储和计算，只在展示时转本地时区**。混用本地时间和 UTC 是排查时间线错乱的头号原因。

```python
import time
time.time()          # 当前时间戳（秒）
time.sleep(3)        # 睡眠 3 秒（调试、限流常用）
time.monotonic()     # 单调时钟，测耗时用它（不受系统时间调整影响）
```

## 7. tempfile：别在 /tmp 手拼文件名

```python
import tempfile

# 临时文件（用完自动删）
with tempfile.NamedTemporaryFile("w", suffix=".yaml", delete=True) as f:
    f.write(manifest)
    f.flush()                                     # 确保写进磁盘
    subprocess.run(["kubectl", "apply", "-f", f.name], check=True)

# 临时目录（整个目录用完自动删）
with tempfile.TemporaryDirectory() as d:
    Path(d, "a.txt").write_text("x")

# 只要一个安全的名字
fd, path = tempfile.mkstemp()
os.close(fd)
os.unlink(path)
```

**手拼 `/tmp/fixed_name` 的问题是并发冲突**：两个实例同时跑会互相覆盖。`tempfile` 保证名字唯一。

## 8. signal + atexit：优雅退出

```python
import signal, sys, atexit

def on_signal(signum, frame):
    print(f"收到信号 {signum}，开始清理")
    cleanup()
    sys.exit(0)

signal.signal(signal.SIGTERM, on_signal)   # K8s 停 Pod 时发的就是它
signal.signal(signal.SIGINT, on_signal)    # Ctrl+C

atexit.register(lambda: print("进程正常退出时会执行"))
```

**容器里尤其重要**：K8s 停止 Pod 时先发 SIGTERM，给 `terminationGracePeriodSeconds`（默认 30s）。脚本能接住就能清理临时资源、注销注册；接不住就被 SIGKILL 硬杀。

## 9. 实战：按水位清理日志

```python
#!/usr/bin/env python3
"""删除 7 天前的日志，并打印磁盘使用率。"""
import argparse, logging, time
from pathlib import Path

log = logging.getLogger("clean")

def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("dir", help="日志目录")
    ap.add_argument("--days", type=int, default=7)
    ap.add_argument("--dry-run", action="store_true")
    args = ap.parse_args()

    logging.basicConfig(level=logging.INFO,
                        format="%(asctime)s %(levelname)s %(message)s")

    cutoff = time.time() - args.days * 86400
    removed = freed = 0

    for f in Path(args.dir).rglob("*.log"):
        if f.stat().st_mtime >= cutoff:
            continue
        size = f.stat().st_size
        if args.dry_run:
            log.info("[dry-run] 将删除 %s (%.1f MB)", f, size / 1024**2)
        else:
            f.unlink()
            log.info("已删除 %s (%.1f MB)", f, size / 1024**2)
        removed += 1
        freed += size

    total, used, free = shutil.disk_usage(args.dir)
    log.info("共 %d 个文件，释放 %.1f MB；当前使用率 %.1f%%",
             removed, freed / 1024**2, used / total * 100)
    return 0

if __name__ == "__main__":
    raise SystemExit(main())
```

## 核心概念

1. **`subprocess` 用列表传参 + 必设 `timeout` + `text=True`**，`shell=True` 慎用。
2. **`argparse` 是脚本参数的标准做法**，别用 `input()`。
3. **日志用 `logging` 不用 `print`**，`%s` 占位是惰性求值。
4. **时间统一用 UTC**，展示层再转本地。
5. **临时文件用 `tempfile`**，手拼路径会并发冲突。
6. **`shutil.disk_usage` 算水位**，`shutil.rmtree(ignore_errors=True)` 做清理。
7. **长驻脚本要处理 SIGTERM**，否则容器停不掉、资源漏清理。
