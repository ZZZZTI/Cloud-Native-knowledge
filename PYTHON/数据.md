### 变量

```python
# Python 不需要提前声明类型，直接「名字 = 值」就能创建变量。

# ---------- 1. 变量：给数据起个名字 ----------
name = "小明"          # 字符串（文本），用引号包起来
age = 18               # 整数
height = 1.75          # 浮点数（小数）
is_student = True      # 布尔值，只有 True / False 两个取值

print("姓名:", name)
print("年龄:", age)
print("身高:", height)
print("是学生吗:", is_student)

# ---------- 2. 查看类型：type() ----------
print("\n--- 它们的类型 ---")
print(type(name))        # <class 'str'>
print(type(age))         # <class 'int'>
print(type(height))      # <class 'float'>
print(type(is_student))  # <class 'bool'>

# ---------- 3. f-string：最常用的字符串拼接方式 ----------
message = f"{name} 今年 {age} 岁，身高 {height} 米。"
print("\n--- f-string ---")
print(message)

# 还能在 {} 里直接做运算、控制小数位数
print(f"明年他就 {age + 1} 岁了")
print(f"身高保留一位小数：{height:.1f}")

# ---------- 4. 数字运算 ----------
print("\n--- 数字运算 ---")
a, b = 7, 2
print("a + b =", a + b)   # 9    加
print("a - b =", a - b)   # 5    减
print("a * b =", a * b)   # 14   乘
print("a / b =", a / b)   # 3.5  除（结果一定是小数）
print("a // b =", a // b)  # 3   整除，只要整数部分
print("a % b =", a % b)   # 1    取余数
print("a ** b =", a ** b)  # 49   a 的 b 次方

# ---------- 5. 字符串常用操作 ----------
print("\n--- 字符串 ---")
text = "  Hello Python  "
print("原样        :", repr(text))
print("去掉两边空格:", repr(text.strip()))
print("全大写      :", text.upper())
print("替换        :", text.replace("Python", "世界"))
print("长度        :", len(text))
print("切片取前5个 :", text[2:7])

# ---------- 6. 输入：让程序跟人对话 ----------
# input() 会等你在终端里打字并回车，得到的结果永远是「字符串」。
user = input("\n请输入你的名字：")
print(f"你好，{user}！欢迎开始学 Python。")

# 如果要输入数字，记得转换类型：
num = int(input("请输入一个整数："))
print(f"它的两倍是 {num * 2}")

# 三种最常用的转换
port_str = "80"
port_int = int(port_str)        # 字符串 -> 整数，结果 80
print(port_int + 1)             # 81，能算了

ratio = float("0.95")           # 字符串 -> 小数，结果 0.95

num = 42
num_str = str(num)              # 整数 -> 字符串，结果 "42"

# 反过来，数字和字符串相加会直接报错：
# print("端口: " + 80)          # TypeError！字符串不能直接加数字
print("端口: " + str(80))       # 正确
print(f"端口: {80}")            # 更好，f-string 自动转换
```

### 容器

```py
"""第 4 课：列表、字典、集合、元组。

这四种「容器」用来把一堆数据装在一起，是 Python 最常用的工具。
"""

# ---------- 1. 列表 list：有序、可修改 ----------
print("--- 列表 list ---")
scores = [88, 92, 79, 95, 60]

print("全部成绩:", scores)
print("第一个  :", scores[0])     # 下标从 0 开始
print("最后一个:", scores[-1])    # -1 表示倒数第一个
print("前三个  :", scores[:3])    # 切片，不含下标 3
print("有几个  :", len(scores))
print("最高分  :", max(scores), " 最低分:", min(scores))
print("总分    :", sum(scores))
print("平均分  :", sum(scores) / len(scores))

scores.append(100)             # 末尾加一个
scores.insert(0, 50)           # 在下标 0 处插入
scores.remove(50)              # 删掉值为 50 的那个
scores.sort(reverse=True)      # 从大到小排序
print("整理后  :", scores)

# 创建
ports = [80, 443, 3306, 6379]
hosts = ["10.0.0.1", "10.0.0.2"]
mixed = ["nginx", 80, True]        # 可以混类型，但实际很少这么用
empty = []                          # 空列表，后面往里面加

# 索引：从 0 开始数！
print(ports[0])        # 80   （第一个）
print(ports[1])        # 443  （第二个）
print(ports[-1])       # 6379 （最后一个，-1 表示倒数第一）
print(len(ports))      # 4    （长度，很常用）

# 增删
ports.append(5432)          # 末尾加一个 -> [80, 443, 3306, 6379, 5432]
ports.remove(443)           # 删掉 443
print(ports)                # [80, 3306, 6379, 5432]

# 判断在不在里面（超常用）
print(80 in ports)          # True
print(22 in ports)          # False


# ---------- 2. 字典 dict：键值对，按名字取值 ----------
print("\n--- 字典 dict ---")
student = {
    "name": "小明",
    "age": 18,
    "major": "计算机",
    "scores": [88, 92, 79],
}

print("姓名:", student["name"])          # 用键取值
print("专业:", student.get("major"))
print("不存在的键用 get 不会报错:", student.get("phone", "没有填写"))

student["email"] = "xiaoming@example.com"   # 新增键值对
student["age"] = 19                          # 修改
print("所有键:", list(student.keys()))
print("所有值:", list(student.values()))

# 遍历字典（最常用的写法）
print("逐项打印：")
for key, value in student.items():
    print(f"  {key} → {value}")

    service = {
    "name": "nginx",
    "port": 80,
    "cpu": 12.5,
    "running": True
}

# 取值：用键取
print(service["name"])          # nginx
print(service["port"])          # 80

# 安全取值：get，键不存在时返回默认值，不报错
print(service.get("name"))              # nginx
print(service.get("memory", "未知"))     # 未知（因为没有 memory 这个键）

# 改值 / 加值
service["cpu"] = 15.0                   # 改
service["memory"] = "2G"                # 加
print(service)

# 判断键在不在
print("port" in service)        # True
print("disk" in service)        # False


# ---------- 3. 集合 set：自动去重、无序 ----------
print("\n--- 集合 set ---")
numbers = [1, 2, 2, 3, 3, 3, 4]
unique = set(numbers)
print("原列表:", numbers)
print("去重后:", unique)

a = {1, 2, 3, 4}
b = {3, 4, 5, 6}
print("交集:", a & b)     # 两边都有的
print("并集:", a | b)     # 合在一起
print("差集:", a - b)     # a 有而 b 没有的


# ---------- 4. 元组 tuple：有序、不可修改 ----------
print("\n--- 元组 tuple ---")
point = (3, 5)
x, y = point             # 解包
print(f"坐标 x={x}, y={y}")
# point[0] = 10          # 取消注释会报错，元组创建后不能改

# ---------- 5. 推导式：一行生成一个容器（Python 特色，非常好用）----------
print("\n--- 推导式 ---")
print("1~10 的平方:", [n ** 2 for n in range(1, 11)])
print("其中的偶数  :", [n for n in range(1, 11) if n % 2 == 0])
print("字符串长度  :", {w: len(w) for w in ["apple", "hi", "python"]})

# ---------- 6. 综合例子：统计词频 ----------
print("\n--- 词频统计 ---")
sentence = "the quick brown fox jumps over the lazy dog the fox"
word_count = {}
for word in sentence.split():        # split() 按空格切开成列表
    word_count[word] = word_count.get(word, 0) + 1

# 按出现次数从多到少排序
for word, count in sorted(word_count.items(), key=lambda item: -item[1]):
    print(f"  {word}: {count}")
```
