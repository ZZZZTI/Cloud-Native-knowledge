## 1. 读写文本文件

```python
# 读：用 with，结束自动关闭（异常时也会关）
with open("hosts.txt", encoding="utf-8") as f:
    content = f.read()             # 一次读完，得到一个大字符串

with open("hosts.txt", encoding="utf-8") as f:
    lines = f.readlines()          # 读成列表，每行带 \n

# 最推荐：逐行迭代，不占内存
with open("hosts.txt", encoding="utf-8") as f:
    for line in f:
        line = line.strip()        # ⚠️ 一定要 strip，否则带 \n
        if not line or line.startswith("#"):
            continue               # 跳过空行和注释行
        print(line)

# 写：会覆盖原文件！
with open("out.txt", "w", encoding="utf-8") as f:
    f.write("hello\n")

# 追加
with open("out.txt", "a", encoding="utf-8") as f:
    f.write("world\n")
```

| 模式 | 含义 | 注意 |
| :-- | :-- | :-- |
| `"r"` | 只读（默认） | 文件不存在报错 |
| `"w"` | 只写 | **文件已存在会被清空** |
| `"a"` | 追加 | 不存在则创建 |
| `"x"` | 独占创建 | 已存在则报错（防止误覆盖） |
| `"rb"/"wb"` | 二进制 | 图片、压缩包 |

## 2. pathlib：路径操作首选

```python
from pathlib import Path

p = Path("/var/log/nginx")
p / "access.log"                 # 拼接：/var/log/nginx/access.log
p.exists(); p.is_dir(); p.is_file()
p.name                           # 'nginx'
p.suffix                         # ''
p.stem                           # 'nginx'
p.parent                         # /var/log
p.stat().st_size                 # 字节数

# 遍历
for f in p.glob("*.log"):        # 当前目录下的 log
    print(f.name)

for f in p.rglob("*.log"):       # 递归
    print(f, f.stat().st_size)

# 直接读写（短文件很方便）
Path("a.txt").read_text(encoding="utf-8")
Path("b.txt").write_text("done\n", encoding="utf-8")

# 创建 / 删除
Path("/tmp/x/y").mkdir(parents=True, exist_ok=True)
Path("a.txt").unlink(missing_ok=True)
```

用 `pathlib` 而不是字符串拼接，能避免 `/` 重复或漏掉的低级错误。

## 3. JSON：API 和配置的标准格式

```python
import json

data = {"name": "web", "replicas": 3, "ports": [80, 443]}

# 字典 → 字符串
json.dumps(data)
json.dumps(data, ensure_ascii=False, indent=2)   # 中文不转义 + 美化（写给人看必用）
json.dumps(data, sort_keys=True)                 # key 排序，便于 diff

# 字符串 → 字典
json.loads('{"a": 1}')

# 文件读写
with open("deploy.json", encoding="utf-8") as f:
    data = json.load(f)

with open("out.json", "w", encoding="utf-8") as f:
    json.dump(data, f, ensure_ascii=False, indent=2)
```

⚠️ **`ensure_ascii=False` 必加**，否则中文会变成 `\u4e2d\u6587`，日志和配置完全没法看。

⚠️ **解析外部数据要兜底**：

```python
try:
    data = json.loads(resp_text)
except json.JSONDecodeError as e:
    print("返回的不是合法 JSON：", resp_text[:200])
    data = {}
```

## 4. YAML：K8s 和 Ansible 的语言

```python
import yaml

with open("deploy.yaml", encoding="utf-8") as f:
    doc = yaml.safe_load(f)

# 多文档（用 --- 分隔的文件）要用 load_all
with open("all.yaml", encoding="utf-8") as f:
    docs = list(yaml.safe_load_all(f))

# 写回
yaml.safe_dump(doc, allow_unicode=True, sort_keys=False)
# allow_unicode=True 让中文正常输出；sort_keys=False 保持原顺序
```

⚠️ **只用 `safe_load`**。`yaml.load()` 不带 Loader 时可以构造任意 Python 对象，解析不可信内容有安全风险。

```python
yaml.safe_load("a: 1")        # ✅
yaml.load("a: 1")             # ❌ 不安全，且新版要求显式传 Loader
```

## 5. CSV：报表和导出

```python
import csv

# 读：用 DictReader，按列名取值，比按下标稳
with open("metrics.csv", encoding="utf-8", newline="") as f:
    for row in csv.DictReader(f):
        print(row["host"], row["cpu"])

# 写
rows = [{"host": "k8s-01", "cpu": "8"}, {"host": "k8s-02", "cpu": "16"}]
with open("out.csv", "w", encoding="utf-8", newline="") as f:
    writer = csv.DictWriter(f, fieldnames=["host", "cpu"])
    writer.writeheader()
    writer.writerows(rows)
```

`newline=""` 不能省，否则 CSV 里会多出空行。

## 6. 原子写入：防止写一半被读

```python
import os, tempfile

def atomic_write(path: str, content: str) -> None:
    """先写临时文件，再原子改名，避免读方看到半个文件。"""
    d = os.path.dirname(os.path.abspath(path))
    fd, tmp = tempfile.mkstemp(dir=d)
    try:
        with os.fdopen(fd, "w", encoding="utf-8") as f:
            f.write(content)
            f.flush()
            os.fsync(f.fileno())      # 确保落盘
        os.replace(tmp, path)         # 原子替换
    except Exception:
        os.unlink(tmp)                # 失败清理临时文件
        raise
```

**什么时候必须用**：改配置文件（如 nginx.conf、kubeconfig）。直接 `open(w)` 会先把文件清空，此刻如果有进程读，就拿到空文件。

## 7. 大文件处理

```python
# ❌ 一次读进内存，1GB 日志直接把机器拖死
content = open("big.log", encoding="utf-8").read()

# ✅ 逐行迭代，内存占用恒定
with open("big.log", encoding="utf-8", errors="replace") as f:
    for line in f:
        if "ERROR" in line:
            print(line.rstrip())
```

```python
# 按块读二进制大文件
def sha256_of(path: str) -> str:
    import hashlib
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):   # 1MB 一块
            h.update(chunk)
    return h.hexdigest()
```

## 8. 实战：批量给 K8s YAML 换镜像 tag

```python
import yaml
from pathlib import Path

NEW_TAG = "v1.2.3"

for f in Path("manifests").rglob("*.yaml"):
    docs = list(yaml.safe_load_all(f.read_text(encoding="utf-8")))

    changed = False
    for d in docs:
        if not isinstance(d, dict) or d.get("kind") != "Deployment":
            continue
        for c in d["spec"]["template"]["spec"]["containers"]:
            image = c["image"]
            name, _, _old = image.rpartition(":")   # 去掉旧 tag
            c["image"] = f"{name}:{NEW_TAG}"
            changed = True

    if changed:
        # 原子写回，避免别人读到时文件不完整
        tmp = f.with_suffix(".yaml.tmp")
        tmp.write_text(
            yaml.safe_dump_all(docs, allow_unicode=True, sort_keys=False),
            encoding="utf-8",
        )
        tmp.replace(f)
        print("已更新", f)
```

## 核心概念

1. **永远 `with open(...)` + `encoding="utf-8"`**，两个都不能省。
2. **`"w"` 模式会清空文件**，改配置用「临时文件 + `os.replace`」原子写。
3. **逐行迭代处理大文件**，不要 `read()` 全读进内存。
4. **`pathlib` 代替字符串拼路径**，天然不会丢分隔符。
5. **`json.dumps(..., ensure_ascii=False)`**，中文才正常显示。
6. **YAML 只用 `safe_load`**，`load_all` 读多文档。
7. **CSV 用 `DictReader`/`DictWriter`**，按列名读写，加 `newline=""`。
