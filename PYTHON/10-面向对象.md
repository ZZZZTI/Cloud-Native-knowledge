## 1. 为什么运维需要类

```python
# 不用类：参数在函数之间传来传去，越传越多
def list_pods(token, base_url, namespace, timeout): ...
def get_logs(token, base_url, namespace, pod, timeout): ...
# 每个函数都要重复传 token/base_url，漏一个就出错

# 用类：把「共同的状态」收进对象里
client = K8sClient(token=..., base_url=..., timeout=10)
client.list_pods("prod")
client.get_logs("prod", "web-1")
```

**判断标准**：如果一组函数总在传同样的几个参数，就该封成类。

## 2. 最小的类

```python
class K8sClient:
    # __init__ 是构造方法，创建对象时自动调用
    # self 代表「这个对象本身」，必须是第一个参数（名字可以改，但没人改）
    def __init__(self, token: str, base_url: str, timeout: int = 10):
        # 把参数存到 self 上，成为「实例属性」
        self.token = token
        self.base_url = base_url
        self.timeout = timeout

    # 普通方法，第一个参数也是 self
    def list_pods(self, namespace: str) -> list:
        print(f"GET {self.base_url}/pods?ns={namespace}")
        return []

# 创建实例（注意：不需要传 self）
c = K8sClient(token="abc", base_url="https://api")
c.list_pods("prod")        # 调用时 self 自动绑定成 c
```

## 3. 实例属性 vs 类属性

```python
class Node:
    kind = "node"                 # 类属性：所有实例共享

    def __init__(self, name):
        self.name = name          # 实例属性：每个对象独立

a, b = Node("n1"), Node("n2")
a.name        # 'n1'
b.name        # 'n2'
a.kind        # 'node'
Node.kind     # 'node'

# ⚠️ 可变类属性是陷阱
class Bad:
    items = []                   # ❌ 所有实例共享同一个列表
    def add(self, x):
        self.items.append(x)

Bad().add(1); Bad().items        # [1]  ← 另一个实例也看到了

# ✅ 可变状态一律放实例属性
class Good:
    def __init__(self):
        self.items = []
```

## 4. 继承：复用共同部分

```python
class BaseChecker:
    """所有检查项的基类。"""

    def __init__(self, name: str):
        self.name = name

    def check(self) -> bool:
        """子类必须实现这个方法。"""
        raise NotImplementedError

class PodChecker(BaseChecker):
    def check(self) -> bool:
        print(f"{self.name}: 检查 Pod")
        return True

class NodeChecker(BaseChecker):
    def check(self) -> bool:
        print(f"{self.name}: 检查节点")
        return True

# 统一调用，不用关心具体类型
for checker in [PodChecker("pods"), NodeChecker("nodes")]:
    checker.check()
```

运维里继承用得不多，但这个模式很实用：**一组检查项有统一接口**，主流程直接遍历调用。

```python
# 需要复用父类实现时用 super()
class Prober(BaseChecker):
    def __init__(self, name, timeout=5):
        super().__init__(name)        # 调用父类构造函数
        self.timeout = timeout
```

## 5. 魔术方法：让对象「像内置类型」

```python
class Result:
    def __init__(self, name, ok, cost):
        self.name, self.ok, self.cost = name, ok, cost

    def __repr__(self):
        """调试时打印对象的样子（开发者看）"""
        return f"Result(name={self.name!r}, ok={self.ok}, cost={self.cost})"

    def __str__(self):
        """print() 时显示的内容（用户看）"""
        return f"{self.name}: {'OK' if self.ok else 'FAIL'} ({self.cost:.2f}s)"

    def __eq__(self, other):
        """决定 == 怎么比较"""
        return isinstance(other, Result) and self.name == other.name

    def __len__(self):
        """决定 len(obj) 返回什么"""
        return 1

r = Result("api", True, 0.12)
print(r)          # api: OK (0.12s)      ← __str__
r                 # Result(name='api', ...)  ← __repr__
```

最常用的是 `__repr__`：调试时直接 `print(obj)` 就能看到全部字段，比 `<object at 0x10f...>` 有用一万倍。

## 6. @property：把方法当属性用

```python
class Disk:
    def __init__(self, total, used):
        self.total = total
        self.used = used

    @property
    def percent(self):
        """访问时不用加括号：disk.percent"""
        return self.used / self.total * 100

d = Disk(100, 85)
print(d.percent)      # 85.0
d.percent = 90        # ❌ 只读，会报 AttributeError
```

```python
# 带 setter：赋值时做校验
class Port:
    def __init__(self, value):
        self._value = value

    @property
    def value(self):
        return self._value

    @value.setter
    def value(self, v):
        if not 1 <= v <= 65535:
            raise ValueError(f"非法端口: {v}")
        self._value = v
```

## 7. dataclass：只装数据的类

```python
from dataclasses import dataclass, field

@dataclass
class Pod:
    name: str
    namespace: str = "default"
    restarts: int = 0
    labels: list[str] = field(default_factory=list)   # 可变默认值必须用 factory

p = Pod("web-1", restarts=3)
print(p)        # Pod(name='web-1', namespace='default', restarts=3, labels=[])
```

`@dataclass` 自动生成 `__init__`、`__repr__`、`__eq__`，省掉一堆样板代码。**纯数据结构优先用它，别手写 `__init__`**。

## 8. 上下文管理器：with 背后是什么

```python
class Timer:
    """用法：with Timer('拉取 Pod'): ... 自动打印耗时。"""

    def __init__(self, label):
        self.label = label

    def __enter__(self):
        """进入 with 时执行，返回值赋给 as 后面的变量。"""
        self.start = time.time()
        return self

    def __exit__(self, exc_type, exc, tb):
        """退出 with 时执行，即使发生异常也会走。"""
        cost = time.time() - self.start
        print(f"{self.label} 耗时 {cost:.3f}s")
        return False        # 返回 True 会吞掉异常，通常返回 False

with Timer("拉取 Pod"):
    pods = fetch_pods()
```

`with open(...)` 就是这个机制：`__enter__` 打开文件，`__exit__` 负责关闭（所以异常时也会关）。

## 9. 实战：封装一个带重试的 API 客户端

```python
import time, requests

class ApiError(Exception): ...

class ApiClient:
    """带鉴权、超时、重试、统计的 HTTP 客户端。"""

    def __init__(self, base_url: str, token: str, timeout: int = 10, retries: int = 3):
        self.base_url = base_url.rstrip("/")
        self.timeout = timeout
        self.retries = retries
        self.calls = 0                      # 实例属性：统计调用次数
        self.session = requests.Session()
        self.session.headers.update({"Authorization": f"Bearer {token}"})

    def _request(self, method: str, path: str, **kwargs) -> dict:
        """内部方法（下划线开头表示不对外）。"""
        url = f"{self.base_url}{path}"
        last_err = None

        for attempt in range(1, self.retries + 1):
            self.calls += 1
            try:
                r = self.session.request(method, url, timeout=self.timeout, **kwargs)
                if r.status_code >= 500:          # 服务端错误，可重试
                    raise ApiError(f"{r.status_code} {r.text[:100]}")
                r.raise_for_status()
                return r.json()
            except (requests.RequestException, ApiError) as e:
                last_err = e
                if attempt == self.retries:
                    break
                wait = 2 ** (attempt - 1)          # 指数退避 1s,2s,4s
                print(f"第 {attempt} 次失败，{wait}s 后重试: {e}")
                time.sleep(wait)

        raise ApiError(f"请求失败（已重试 {self.retries} 次）: {last_err}")

    def get_pods(self, namespace: str) -> list[dict]:
        return self._request("GET", "/api/v1/pods", params={"namespace": namespace}).get("items", [])

    def close(self):
        self.session.close()

    # 支持 with 用法
    def __enter__(self):
        return self

    def __exit__(self, *exc):
        self.close()

# 使用
with ApiClient("https://api.example.com", token="xxx") as c:
    pods = c.get_pods("prod")
print("总调用次数:", c.calls)
```

这就是运维里类的典型用法：**把「连接、鉴权、超时、重试、统计」这些横切逻辑收进对象**，业务方法只关心「取什么数据」。

## 核心概念

1. **一组建函数总在传同样的参数 → 封成类**，这是运维用 OOP 的唯一判断标准。
2. **`self` 是对象本身**，实例属性写在 `__init__` 里。
3. **可变状态别放类属性**，会所有实例共享。
4. **`__repr__` 最值得写**，调试时 `print(obj)` 直接看到内容。
5. **`@dataclass` 省样板代码**，纯数据类优先用它。
6. **`@property` 让计算属性像字段一样访问**，还能加赋值校验。
7. **`__enter__`/`__exit__` 就是 `with` 的原理**，用完自动清理资源。
8. **下划线开头表示内部方法**，不是语法强制，但是团队约定。
