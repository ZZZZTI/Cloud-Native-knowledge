### 逻辑

```Shell
"""第 2 课：条件判断与循环。

Python 用「缩进」来表示代码块，同一层级的缩进必须一致（约定用 4 个空格）。
"""

# 比较，结果永远是 True 或 False
print(80 > 50)      # True
print(80 == 80)     # True  （注意是两个等号）
print(80 != 81)     # True
print(80 <= 79)     # False

# 逻辑
print(True and False)   # False（两个都真才真）
print(True or False)    # True （一个真就真）
print(not True)         # False


# ---------- 1. if / elif / else ----------
print("--- 判断成绩等级 ---")
score = 86  # 改这个数字，观察输出变化

if score >= 90:
    print("优秀")
elif score >= 80:
    print("良好")
elif score >= 60:
    print("及格")
else:
    print("需要努力")

# ---------- 2. 比较与逻辑运算 ----------
print("\n--- 逻辑运算 ---")
age = 20
has_ticket = True

# and：两个都成立；or：有一个成立；not：取反
if age >= 18 and has_ticket:
    print("可以入场")
else:
    print("不能入场")

print("10 == 10 :", 10 == 10)     # 相等
print("10 != 10 :", 10 != 10)     # 不相等
print("3 in [1, 2, 3] :", 3 in [1, 2, 3])   # 在……里面

# 三元表达式：一行写完简单的 if-else
status = "成年" if age >= 18 else "未成年"
print("状态:", status)

# ---------- 3. for 循环：遍历一串东西 ----------
print("\n--- for 循环 ---")
fruits = ["苹果", "香蕉", "橙子"]

for fruit in fruits:
    print("我喜欢吃", fruit)

# range(n) 生成 0 到 n-1 的整数
print("\n1 到 5 的平方：")
for i in range(1, 6):
    print(f"{i} 的平方是 {i ** 2}")

# enumerate 同时拿到「下标 + 元素」
print("\n带序号的列表：")
for index, fruit in enumerate(fruits, start=1):
    print(f"{index}. {fruit}")

# ---------- 4. while 循环：条件成立就一直转 ----------
print("\n--- while 循环 ---")
count = 3
while count > 0:
    print(f"倒计时 {count}")
    count -= 1        # 等价于 count = count - 1，千万别忘了这行，否则死循环
print("发射！🚀")

# ---------- 5. break 和 continue ----------
print("\n--- break 和 continue ---")
for i in range(1, 11):
    if i == 3:
        continue      # 跳过本次循环剩下的部分，直接进入下一次
    if i == 7:
        break         # 彻底结束整个循环
    print(i, end=" ")
print()

# ---------- 6. 循环套循环：九九乘法表 ----------
print("\n--- 九九乘法表 ---")
for i in range(1, 10):
    for j in range(1, i + 1):
        print(f"{j}×{i}={i * j}", end="\t")  # end="\t" 表示用制表符代替换行
    print()  # 内层循环结束后换行

# ---------- 小练习 ----------
# 1) 用 for 循环算出 1 到 100 的总和。
# 2) 遍历 1 到 20，只打印偶数（提示：if i % 2 == 0）。
# 3) 把上面 score 改成 45，确认输出变成「需要努力」。


```

### 函数

```Shell
"""第 3 课：函数。

函数 = 把一段代码打包起来，起个名字，之后随时调用。
好处：不用重复写、逻辑更清晰、方便修改。
"""

# ---------- 1. 最简单的函数 ----------
def say_hello():
    """打印一句问候。（三个引号里的这段叫文档字符串，说明函数干什么用）"""
    print("你好，欢迎学习 Python！")


say_hello()   # 调用函数：写函数名 + 括号
say_hello()   # 可以调用任意多次


# ---------- 2. 带参数的函数 ----------
def greet(name):
    """根据传入的名字打招呼。name 是「参数」，调用时传入的值叫「实参」。"""
    print(f"你好，{name}！")


greet("小红")
greet("小刚")


# ---------- 3. 带返回值的函数 ----------
def add(a, b):
    """返回 a + b 的结果。"""
    return a + b          # return 把结果交回给调用者


result = add(3, 5)
print(f"add(3, 5) = {result}")
print(f"add(10, 20) * 2 = {add(10, 20) * 2}")


# ---------- 4. 默认参数 ----------
def power(base, exponent=2):
    """求 base 的 exponent 次方，exponent 不给就默认平方。"""
    return base ** exponent


print("power(5)    =", power(5))        # 5 的平方
print("power(5, 3) =", power(5, 3))     # 5 的立方


# ---------- 5. 多个返回值（其实是返回了一个元组）----------
def min_and_max(numbers):
    """同时返回最小值和最大值。"""
    return min(numbers), max(numbers)


low, high = min_and_max([3, 1, 4, 1, 5, 9, 2])
print(f"最小值 {low}，最大值 {high}")


# ---------- 6. 综合小例子：判断素数 ----------
def is_prime(n):
    """判断 n 是不是素数（质数）。"""
    if n < 2:
        return False
    # 只需要试到 n 的平方根就够了
    for i in range(2, int(n ** 0.5) + 1):
        if n % i == 0:
            return False
    return True


print("\n50 以内的素数：", [n for n in range(2, 51) if is_prime(n)])


# ---------- 7. 作用域：函数内部的变量外面看不见 ----------
def demo_scope():
    inner = "我在函数里面"
    print(inner)


demo_scope()
# print(inner)   # 取消注释会报 NameError，因为 inner 只在函数内部存在


# ---------- 小练习 ----------
# 1) 写一个函数 area(width, height) 返回长方形面积，并打印 area(4, 5)。
# 2) 写一个函数 celsius_to_fahrenheit(c) 把摄氏度转成华氏度（公式：c * 9/5 + 32）。
# 3) 写一个函数 count_vowels(text) 数出一个字符串里有多少个元音字母 aeiou。
```

### 对象

```Shell
"""第 5 课：类与对象，以及模块导入。

类（class）= 自己定义一种「数据类型」，把数据和操作数据的函数打包在一起。
对象（object / 实例）= 用这个类型造出来的具体东西。
"""

# ---------- 1. 定义一个类 ----------
class Dog:
    """一只狗。"""

    # __init__ 是「构造方法」，创建对象时自动调用，用来初始化属性
    def __init__(self, name, age):
        self.name = name      # self 表示「这只狗自己」
        self.age = age

    # 类里面的函数叫「方法」，第一个参数永远是 self
    def bark(self):
        print(f"{self.name} 汪！汪！")

    def birthday(self):
        """过生日，年龄加一。"""
        self.age += 1
        print(f"{self.name} 现在 {self.age} 岁了")


# ---------- 2. 创建对象并使用 ----------
print("--- 对象的基本使用 ---")
wang = Dog("旺财", 3)      # 这就创建了一个 Dog 对象
wang.bark()
print(f"{wang.name} 今年 {wang.age} 岁")
wang.birthday()


# ---------- 3. 继承：在原有类的基础上扩展 ----------
class GuideDog(Dog):
    """导盲犬，它也是一种狗，但多了一个本领。"""

    def __init__(self, name, age, owner):
        super().__init__(name, age)   # 先让父类把名字年龄处理好
        self.owner = owner

    def guide(self):
        print(f"{self.name} 正在引导主人 {self.owner} 过马路")


print("\n--- 继承 ---")
lucky = GuideDog("Lucky", 2, "张阿姨")
lucky.bark()       # 从 Dog 继承来的方法，直接能用
lucky.guide()      # 自己新增的方法


# ---------- 4. 特殊方法：让对象打印起来更好看 ----------
class Book:
    def __init__(self, title, author):
        self.title = title
        self.author = author

    def __str__(self):
        """定义 print(对象) 时显示什么。"""
        return f"《{self.title}》— {self.author}"


print("\n--- 特殊方法 __str__ ---")
print(Book("流畅的 Python", "Luciano Ramalho"))


# ---------- 5. 模块与导入 ----------
print("\n--- 模块导入 ---")
import math                    # 导入标准库

print("16 的平方根:", math.sqrt(16))

from datetime import date      # 只导入需要的部分
today = date.today()
print("今天是:", today)

import random                  # 随机数
print("随机数:", random.randint(1, 100))
```
