## 1. os / sys / pathlib ★★★ —— 路径、环境变量、进程参数、退出码

### 1.1 os：常用属性与方法

```
os.environ               环境变量字典（读配置的主要来源）
os.environ.get(k, d)     取值，缺省返回 d
os.environ[k]            取值，缺失抛 KeyError（必需项用这个）
os.getcwd()              当前工作目录
os.chdir(p)              切换工作目录（少用，容易埋 bug）
os.getpid() / getppid()  本进程 / 父进程 PID
os.cpu_count()           逻辑 CPU 核数
os.getloadavg()          1/5/15 分钟负载（Unix）
os.uname()               内核信息（Unix，等同于 uname -a）
os.listdir(p)            列出目录条目（返回 str 列表）
os.walk(p)               递归遍历，产出 (当前目录, 子目录列表, 文件列表)
os.makedirs(p, exist_ok=True)   递归建目录，已存在不报错
os.makedirs(p, mode=0o755)      指定权限
os.remove(p) / os.unlink(p)     删文件
os.rmdir(p)                     删空目录
os.rename(a, b)                 重命名
os.chmod(p, 0o644)              改权限
os.stat(p)                      取元数据（st_size/st_mtime/st_mode/st_uid）
os.getuid()                     当前用户 UID（Unix）
os.sep / os.linesep / os.pathsep   路径分隔符 / 换行符 / PATH 分隔符
```

```Python
import os

# ==== 环境变量：必需项 vs 可选项 ====
token = os.environ["API_TOKEN"]                       # 缺失就 KeyError，早失败
timeout = int(os.environ.get("API_TIMEOUT", "10"))    # 可选项给默认值

# ==== 递归遍历目录树 ====
# os.walk 是生成器，大目录也不会一次占满内存
for root, dirs, files in os.walk("/var/log"):
    # dirs 可以被原地修改，用来「剪枝」
    dirs[:] = [d for d in dirs if not d.startswith(".")]   # 跳过隐藏目录
    for name in files:
        if name.endswith(".log"):
            full = os.path.join(root, name)
            size = os.stat(full).st_size
            print(f"{full}  {size / 1024**2:.1f} MB")
```

### 1.3 pathlib：路径操作首选

```
Path(p)                  构造路径对象
p / "sub"                拼接（比 os.path.join 干净）
p.parent                 上一级
p.parents[0] / p.parents[1]  逐级向上
p.name                   文件名（含后缀）
p.stem                   文件名（不含最后一级后缀）
p.suffix                 最后一个后缀，如 '.log'
p.suffixes               所有后缀列表，如 ['.tar', '.gz']
p.parts                  路径各部分元组
p.exists()               存在吗
p.is_file() / is_dir()  是文件 / 目录吗
p.is_symlink()           是软链吗
p.stat()                 元数据（st_size / st_mtime）
p.glob("*.log")          当前层模式匹配
p.rglob("*.log")         递归模式匹配
p.iterdir()              遍历直接子项
p.mkdir(parents=True, exist_ok=True)   建目录
p.unlink(missing_ok=True)              删文件
p.rename(new) / p.replace(new)         改名 / 覆盖式改名
p.resolve()              转绝对路径并解析软链
p.with_suffix(".json")   换后缀
p.with_name("b.txt")     换文件名
p.read_text(encoding="utf-8")          读文本
p.write_text(s, encoding="utf-8")      写文本
p.read_bytes() / write_bytes()         读写二进制
```

```Python
from pathlib import Path

# ==== 遍历并统计大文件 ====
total = 0
for f in Path("/var/log").rglob("*.log"):
    if not f.is_file():          # rglob 可能返回目录，必须判断
        continue
    size = f.stat().st_size
    total += size
    if size > 100 * 1024**2:
        print(f"[大文件] {f}  {size / 1024**2:.0f} MB")
print(f"合计 {total / 1024**3:.2f} GB")

# ==== 批量改名：a.log → a.log.bak ====
for f in Path("/tmp/reports").glob("*.log"):
    f.rename(f.with_suffix(".log.bak"))     # with_suffix 会替换最后一个后缀
    # 想「追加」后缀要这样写：
    # f.rename(f.parent / (f.name + ".bak"))
```

⚠️ 坑：

- `Path("a/b").parent` 是 `Path("a")`，但 `Path("a").parent` 是 `Path(".")`，不是空
- `p.suffix` 对 `a.tar.gz` 只返回 `.gz`，要全后缀用 `p.suffixes`
- `glob("*.log")` 只匹配当前层，**递归必须用 `rglob` 或 `**/*.log`**
- 路径里有 `~` 不会自动展开，要 `Path.home()` 或 `Path(os.path.expanduser(p))`

## 2. subprocess ★★★ —— 安全调用外部命令、捕获输出与返回码

唯一正确的调用外部命令方式。**列表传参 + 必设 timeout + text=True**。

### subprocess.run 常用参数

```
args                    命令列表（推荐）或字符串（配 shell=True）
capture_output=True     等价于 stdout=PIPE, stderr=PIPE
text=True               返回 str 而非 bytes（省去 decode）
timeout=30              超时秒数，超了抛 TimeoutExpired 并杀掉子进程
check=True              非 0 退出码抛 CalledProcessError
cwd="/path"             子进程工作目录
env={...}               环境变量（注意：会整体替换，不是追加）
input="y\n"             往 stdin 喂数据
shell=True              走 /bin/sh，能用管道但必须防注入
```

### 返回值与异常

```
r.returncode             退出码，0 成功
r.stdout / r.stderr      输出（text=True 时是 str）
r.args                   原始命令
subprocess.TimeoutExpired       超时
subprocess.CalledProcessError    check=True 且退出码非 0
```

```Python
import subprocess

# ==== 标准写法：跑完拿结果 ====
# 列表传参：每个参数独立，不会被 shell 解析，空格和特殊字符都安全
r = subprocess.run(
    ["kubectl", "get", "pods", "-o", "json"],
    capture_output=True,
    text=True,
    timeout=30,          # 必设！否则命令挂住脚本就挂住
    check=False,         # 自己判断 returncode，不抛异常
)
if r.returncode != 0:
    # 失败原因通常在 stderr，一定要打出来
    print(f"kubectl 失败({r.returncode}): {r.stderr.strip()[:200]}")
else:
    import json
    pods = json.loads(r.stdout)["items"]

# ==== 只关心成功/失败 ====
r = subprocess.run(["systemctl", "is-active", "nginx"],
                   capture_output=True, text=True, timeout=5)
is_active = r.stdout.strip() == "active"

# ==== 需要重定向/管道时才用 shell=True ====
# ⚠️ 用了 shell=True，参数里绝不能拼用户输入（命令注入）
r = subprocess.run("ps aux | grep -c nginx",
                   shell=True, capture_output=True, text=True, timeout=10)
count = int(r.stdout.strip())

# ==== 环境变量：默认继承父进程，要改就整体传 ====
import os
env = os.environ.copy()          # ✅ 先复制再改，别直接写 os.environ
env["KUBECONFIG"] = "/etc/kube/config"
subprocess.run(["kubectl", "get", "nodes"], env=env, timeout=10, check=True)
```

```Python
# ==== 超时处理 ====
try:
    subprocess.run(["ping", "-c", "10", "10.0.0.1"], timeout=3)
except subprocess.TimeoutExpired:
    print("探测超时")

# ==== 长驻命令：逐行读输出（tail -f 这类）====
p = subprocess.Popen(
    ["journalctl", "-u", "kubelet", "-f"],
    stdout=subprocess.PIPE, text=True, bufsize=1,     # bufsize=1 = 行缓冲
)
try:
    for line in p.stdout:
        print(line.rstrip())
        if "error" in line.lower():
            break
finally:
    p.terminate()          # 先 SIGTERM
    try:
        p.wait(timeout=5)  # 等它优雅退出
    except subprocess.TimeoutExpired:
        p.kill()           # 不退就 SIGKILL
```

⚠️ 坑：

- 用字符串 + 不用 `shell=True` 时，字符串会被当成**一个可执行文件名**，带空格必然失败
- `shell=True` + f-string 拼参数 = 命令注入漏洞，运维脚本被注入等于服务器被接管
- 子进程的 stdout 是**块缓冲**的，交互式场景要 `-u` 或 `bufsize=1`
- `subprocess.run(["sleep","5"], timeout=1)` 之后子进程已被杀，不用自己清理

## 3. shutil / glob / tempfile ★★★ —— 目录树操作、模式匹配、临时文件

### 3.1 shutil：文件与磁盘

```
shutil.copy(src, dst)      复制文件（权限可能不保留）
shutil.copy2(src, dst)     复制并保留元数据（时间戳等）← 一般用这个
shutil.copytree(a, b)      递归复制目录
shutil.copytree(a, b, dirs_exist_ok=True)   目标已存在也允许
shutil.move(a, b)          移动 / 改名（跨文件系统也能用）
shutil.rmtree(p)           递归删除目录（危险，无回收站）
shutil.rmtree(p, ignore_errors=True)        不存在也不报错
shutil.disk_usage(p)       返回 (total, used, free)，单位字节
shutil.which("kubectl")    查命令的绝对路径（相当于 which）
shutil.make_archive(...)   打 tar/zip 包
shutil.unpack_archive(...) 解包
shutil.chown(p, user=, group=)              改属主（Unix）
```

```Python
import shutil

# ==== 磁盘水位巡检（最常用）====
total, used, free = shutil.disk_usage("/data")
pct = used / total * 100
print(f"使用率 {pct:.1f}%  剩余 {free / 1024**3:.1f} GiB")
if pct > 85:
    print("[WARN] 磁盘水位过高")

# ==== 找命令在哪 ====
kubectl = shutil.which("kubectl")
if kubectl is None:
    raise RuntimeError("找不到 kubectl，检查 PATH")
```

### 3.3 tempfile：临时文件

```
tempfile.mkstemp()            返回 (fd, 路径)，自己负责关闭和删除
tempfile.mkdtemp()            建临时目录
tempfile.NamedTemporaryFile() 有名字的临时文件，with 用完自动删
tempfile.TemporaryDirectory() 临时目录，with 用完递归删
tempfile.gettempdir()         系统临时目录路径
```

```Python
import tempfile, os
from pathlib import Path

# ==== 临时文件：喂给外部命令 ====
with tempfile.NamedTemporaryFile("w", suffix=".yaml", delete=True,
                                 encoding="utf-8") as f:
    f.write(manifest)
    f.flush()                      # ✅ 必须 flush，子进程才读得到内容
    subprocess.run(["kubectl", "apply", "-f", f.name], check=True)

# ==== 临时目录：一堆中间产物 ====
with tempfile.TemporaryDirectory() as d:
    Path(d, "a.txt").write_text("x", encoding="utf-8")
# 退出 with 后整个目录被删

# ==== 需要「原子写」时用 mkstemp + os.replace ====
fd, tmp = tempfile.mkstemp(dir="/etc/nginx")
try:
    with os.fdopen(fd, "w", encoding="utf-8") as fp:
        fp.write(new_conf)
        fp.flush()
        os.fsync(fp.fileno())      # 确保落盘
    os.replace(tmp, "/etc/nginx/nginx.conf")    # 原子替换
except Exception:
    os.unlink(tmp)
    raise
```

⚠️ 坑：手拼 `/tmp/fixed_name` 在并发时会互相覆盖，永远用 `tempfile`。

## 4. psutil ★★★ —— CPU/内存/磁盘/网络/进程指标采集

标准库拿不到的「实时指标」全在这里。健康巡检、阈值告警的基石。

### CPU

```
psutil.cpu_percent(interval=1)     整体使用率 %，interval 是采样窗口
psutil.cpu_percent(interval=1, percpu=True)   每核使用率列表
psutil.cpu_count()                 逻辑核数
psutil.cpu_count(logical=False)    物理核数
psutil.cpu_times()                 各状态累计时间（user/system/idle/iowait）
psutil.cpu_times_percent()         各状态占比 %
psutil.cpu_stats()                 上下文切换、中断次数
psutil.getloadavg()                1/5/15 负载
```

### 内存

```
psutil.virtual_memory()   → total / available / percent / used / free / cached
psutil.swap_memory()      → total / used / percent / sin / sout
```

### 磁盘

```
psutil.disk_partitions()     所有分区 → device / mountpoint / fstype / opts
psutil.disk_usage("/")       → total / used / free / percent
psutil.disk_io_counters()    → read_count / write_count / read_bytes / write_bytes
psutil.disk_io_counters(perdisk=True)    按磁盘拆分
```

### 网络

```
psutil.net_io_counters()        → bytes_sent / bytes_recv / packets_sent / errin / errout
psutil.net_io_counters(pernic=True)       按网卡拆分
psutil.net_connections()       当前连接 → fd / family / type / laddr / raddr / status / pid
psutil.net_if_addrs()          网卡地址
psutil.net_if_stats()          网卡状态（isup / speed / mtu）
```

### 进程

```
psutil.pids()                  所有 PID
psutil.process_iter(["pid","name","cpu_percent","memory_percent"])   批量遍历（推荐）
psutil.Process(pid)            → name() / cmdline() / exe() / cwd() / username()
                                 create_time() / status()
                                 cpu_percent(interval=0.1) / memory_info() / memory_percent()
                                 open_files() / connections() / children() / parent()
                                 terminate() / kill() / wait() / is_running()
```

```Python
import psutil

# ==== 1. 一行拿到核心指标 ====
cpu = psutil.cpu_percent(interval=1)          # ⚠️ interval 必须给，否则是上次到现在的平均值
mem = psutil.virtual_memory()
disk = psutil.disk_usage("/")

print(f"CPU {cpu:.1f}%")
print(f"内存 {mem.percent:.1f}%  (可用 {mem.available / 1024**3:.1f} GiB)")
print(f"磁盘 {disk.percent:.1f}%")

# ==== 2. 找出内存占用 Top 5 的进程 ====
procs = []
for p in psutil.process_iter(["pid", "name", "memory_percent", "cpu_percent"]):
    try:
        info = p.info
        if info["memory_percent"]:
            procs.append(info)
    except psutil.NoSuchProcess:
        continue          # 进程在遍历途中退出了，跳过

procs.sort(key=lambda x: x["memory_percent"], reverse=True)
for p in procs[:5]:
    print(f"{p['pid']:>7}  {p['name']:<20} {p['memory_percent']:>5.1f}%")

# ==== 3. 按名字找进程 ====
for p in psutil.process_iter(["pid", "name", "cmdline"]):
    if p.info["name"] == "nginx":
        print(p.info["pid"], " ".join(p.info["cmdline"] or []))

# ==== 4. 端口被谁占用 ====
for conn in psutil.net_connections(kind="inet"):
    if conn.laddr and conn.laddr.port == 8080:
        try:
            print("占用者:", psutil.Process(conn.pid).name())
        except psutil.NoSuchProcess:
            pass

# ==== 5. 网卡流量（算速率要两次采样求差）====
import time
a = psutil.net_io_counters()
time.sleep(1)
b = psutil.net_io_counters()
print(f"下行 {(b.bytes_recv - a.bytes_recv) / 1024:.0f} KB/s")
```

⚠️ 坑：

- `cpu_percent()` 第一次调用返回 0.0（没有上次采样），循环里第一次要丢弃
- `net_connections()` 在 Linux 上读 `/proc`，普通用户可能拿不到别人的连接（pid 为 None）
- 遍历 `process_iter` 必须 try 住 `NoSuchProcess` / `AccessDenied`，否则进程一退出整个脚本就崩
- `disk_usage` 对网络挂载（NFS）可能阻塞，巡检要加超时或用线程

## 5. platform / pwd / grp ★★ —— 系统信息、用户与组

```
platform.system()         'Linux' / 'Darwin' / 'Windows'
platform.release()        内核版本
platform.version()        更细的版本串
platform.machine()        'x86_64' / 'arm64'
platform.node()           主机名（常用来替代 socket.gethostname）
platform.python_version() '3.12.1'
platform.processor()      CPU 型号（不一定有值）

pwd.getpwnam("app")       → pw_uid / pw_gid / pw_dir / pw_shell
pwd.getpwuid(1000)        反查
grp.getgrnam("docker")    → gr_gid / gr_mem（组成员列表）
```

```Python
import platform
from pathlib import Path

# ==== 跨平台分支 ====
if platform.system() == "Linux":
    log_dir = Path("/var/log")
elif platform.system() == "Darwin":
    log_dir = Path("/usr/local/var/log")

# ==== 判断用户是否存在（Unix）====
try:
    import pwd
    info = pwd.getpwnam("app")
    print(f"uid={info.pw_uid} home={info.pw_dir} shell={info.pw_shell}")
except KeyError:
    print("用户不存在")

# ==== 判断用户在不在某个组（给 docker 权限时常见）====
import grp, os
gid = grp.getgrnam("docker").gr_gid
in_group = gid in os.getgroups()
```

⚠️ `pwd` / `grp` 只在 Unix 有，Windows 上 import 就失败，需要 `sys.platform` 判断后再导入。

## 6. watchdog ★★ —— 监控文件系统事件

监控文件变化，用来做「配置改了自动 reload」「日志轮转后通知」。

```
from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler

handler.on_created(event)     文件/目录被创建
handler.on_modified(event)    被修改
handler.on_deleted(event)     被删除
handler.on_moved(event)       被移动（event.src_path / dest_path）

event.src_path                源路径
event.dest_path               目标路径（仅 moved）
event.is_directory            是目录吗

Observer().schedule(handler, path, recursive=True)   注册
observer.start() / stop() / join()                   启停
```

```Python
import time
from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler

class ConfWatcher(FileSystemEventHandler):
    """nginx 配置一变就打印（真实场景可接 reload）。"""

    def on_modified(self, event):
        # 目录的 modified 事件很吵，先过滤掉
        if event.is_directory:
            return
        if not event.src_path.endswith(".conf"):
            return
        print(f"[变更] {event.src_path}")
        # subprocess.run(["nginx", "-s", "reload"], check=True)

observer = Observer()
observer.schedule(ConfWatcher(), "/etc/nginx", recursive=True)
observer.start()
try:
    while True:
        time.sleep(1)      # 主线程别退出
except KeyboardInterrupt:
    observer.stop()
observer.join()
```

⚠️ 编辑器保存时常产生「rename + create + modify」三次事件，处理逻辑要幂等（或加去抖）。

### 检查清单

```
□ 能用 pathlib 完成遍历 + 过滤 + 统计
```
