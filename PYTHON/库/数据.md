## 1. re / hashlib / base64 ★★★/★★ —— 日志解析、文件校验、编码

### re：日志解析

```
re.compile(pattern, flags)          预编译（循环里必须）
    re.IGNORECASE / re.MULTILINE / re.DOTALL / re.ASCII / re.VERBOSE

re.search(p, s)                     全文找第一个 → Match 或 None
re.match(p, s)                      只从开头匹配
re.fullmatch(p, s)                  整串必须完全匹配
re.findall(p, s)                    全部匹配（有分组时只返回分组）
re.finditer(p, s)                   迭代 Match（能拿 group(0)）
re.sub(p, repl, s, count=0)         替换
re.split(p, s)                      切分
```

```Python
import re

# ==== 命名分组：日志字段提取的标配 ====
LOG = re.compile(
    r'(?P<ip>\d+\.\d+\.\d+\.\d+)'
    r'\s+-\s+-\s+\[(?P<time>[^\]]+)\]'
    r'\s+"(?P<method>[A-Z]+)\s+(?P<path>\S+)[^"]*"'
    r'\s+(?P<status>\d{3})'
    r'\s+(?P<size>\d+)'
)

line = ('10.0.0.1 - - [27/Oct/2026:10:00:01 +0800] '
        '"GET /api/x HTTP/1.1" 502 1234')

m = LOG.search(line)
if m:
    d = m.groupdict()
    print(d["status"], d["path"])          # 502 /api/x
    size = int(d["size"])

# ==== 找不到就跳过，别让脚本崩 ====
for line in open("access.log", encoding="utf-8"):
    m = LOG.search(line)
    if not m:
        continue
    if m["status"].startswith("5"):
        print("5xx:", m["path"])

# ==== 用 sub 做脱敏 ====
masked = re.sub(r"(\d{3})\d{4}(\d{4})", r"\1****\2", "13812345678")
```

⚠️ 见第 3 章（字符串与正则）的细节坑：`findall` 有分组返回分组、`\d` 匹配全角、贪婪 vs 非贪婪。

### hashlib：文件校验

```
hashlib.md5(data).hexdigest()          32 位十六进制（⚠️ 已不安全）
hashlib.sha256(data).hexdigest()       64 位（校验首选）
hashlib.new("sha256")                  按名字创建
h.update(chunk)                        增量更新（大文件必用）
h.hexdigest() / h.digest()             十六进制 / 字节
hashlib.file_digest(f, "sha256")       3.11+，一行算完
hashlib.algorithms_available           当前环境支持的所有算法
```

```Python
import hashlib

def sha256_file(path: str) -> str:
    """大文件必须分块读，不能整个读进内存。"""
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):   # 1MB 一块
            h.update(chunk)
    return h.hexdigest()

# 3.11+ 可以一行搞定
with open("pkg.tar.gz", "rb") as f:
    print(hashlib.file_digest(f, "sha256").hexdigest())

# ==== 校验下载包 ====
def verify(path: str, expected: str) -> bool:
    actual = sha256_file(path)
    if actual != expected.lower():
        print(f"校验失败\n  期望 {expected}\n  实际 {actual}")
        return False
    return True
```

⚠️ `md5` / `sha1` 不能用于安全场景（碰撞可构造）。Python 3.9+ 在 FIPS 环境里会拒绝 `md5()`，要显式写 `usedforsecurity=False`。

### base64：编码

```
base64.b64encode(bytes) → bytes      编码（结果还要 decode）
base64.b64decode(str)   → bytes      解码
base64.urlsafe_b64encode / b64decode 用 - 和 _ 代替 + 和 /（URL、JWT）
base64.b85encode / b32encode         其他变体
```

```Python
import base64

# ==== K8s Secret 的值就是 base64 ====
secret = "cGFzc3dvcmQ="
print(base64.b64decode(secret).decode())        # password

# ==== 编码（注意输入必须是 bytes）====
print(base64.b64encode("密码".encode()).decode())

# ==== 生成 Basic Auth 头 ====
token = base64.b64encode(b"user:pass").decode()
headers = {"Authorization": f"Basic {token}"}

# ==== URL 安全（JWT 里常见）====
print(base64.urlsafe_b64encode(b"\xff\xfe").decode())
```

⚠️ base64 **是编码不是加密**，任何人都能解开，不能用来保护密码。

## 2. tarfile / zipfile / gzip ★★★ —— 备份归档、日志压缩

### tarfile

```
tarfile.open(path, mode)     mode: "r" / "r:gz" / "r:bz2" / "r:xz"
                                   "w" / "w:gz" / "w:bz2" / "w:xz"
tar.add(path, arcname=)      加入文件/目录
tar.addfile(info, fileobj)   手工构造条目
tar.extractall(path, filter="data")   解压（⚠️ 新版要 filter）
tar.extract(member, path, filter="data")
tar.getnames() / getmembers()         成员列表
tar.close()
tarfile.is_tarfile(path)             判断是不是 tar
```

```Python
import tarfile
from pathlib import Path

# ==== 打包（备份最常用）====
with tarfile.open("/backup/app-20261027.tar.gz", "w:gz") as tar:
    tar.add("/opt/app", arcname="app")        # arcname 决定解压后的目录名

# ==== 解压 ====
with tarfile.open("/backup/app-20261027.tar.gz", "r:gz") as tar:
    # ⚠️ filter="data" 必须加：阻止路径穿越和特殊文件（安全修复）
    tar.extractall("/opt/restore", filter="data")

# ==== 只解压其中某个文件 ====
with tarfile.open("/backup/app.tar.gz", "r:gz") as tar:
    members = [m for m in tar.getmembers() if m.name.endswith("config.yaml")]
    tar.extractall("/tmp/x", members=members, filter="data")

# ==== 不解压，直接看内容 ====
with tarfile.open("/backup/app.tar.gz", "r:gz") as tar:
    for m in tar.getmembers()[:10]:
        print(f"{m.size:>12}  {m.name}")
```

### zipfile

```
zipfile.ZipFile(path, "r"/"w"/"a", compression=)
    compression: ZIP_STORED（不压缩）/ ZIP_DEFLATED（常用）/ ZIP_BZIP2 / ZIP_LZMA
zf.write(filename, arcname=)        加入文件
zf.writestr(arcname, data)          直接写字符串/bytes
zf.extractall(path) / extract(member, path)
zf.namelist()                       成员名列表
zf.infolist()                       带大小、时间
zf.read(name)                       读某个成员
zipfile.is_zipfile(path)
```

```Python
import zipfile

# ==== 打包：一定要指定 ZIP_DEFLATED 才会压缩 ====
with zipfile.ZipFile("/backup/logs.zip", "w", zipfile.ZIP_DEFLATED) as zf:
    for p in Path("/var/log").rglob("*.log"):
        # arcname 去掉开头的 /，否则解压出绝对路径
        zf.write(p, arcname=p.relative_to("/var/log"))

# ==== 解压 ====
with zipfile.ZipFile("/backup/logs.zip") as zf:
    zf.extractall("/tmp/restore")

# ==== 只读某个文件 ====
with zipfile.ZipFile("/backup/logs.zip") as zf:
    print(zf.read("app.log")[:200])
```

### gzip

```
import gzip
gzip.open(path, "rt"/"wt"/"rb"/"wb")    文本模式要 "t"
gzip.compress(data) / gzip.decompress(b)
```

```Python
import gzip
import shutil

# ==== 压缩单个大文件（流式，不占内存）====
with open("/var/log/big.log", "rb") as src, \
     gzip.open("/var/log/big.log.gz", "wb") as dst:
    shutil.copyfileobj(src, dst, length=1 << 20)

# ==== 读压缩日志 ====
with gzip.open("/var/log/big.log.gz", "rt", encoding="utf-8") as f:
    for line in f:
        if "ERROR" in line:
            print(line.rstrip())
```

⚠️ 坑：

- `tarfile.extractall` 不加 `filter="data"` 在 3.12+ 会警告，3.14 会默认启用；**老版本上是路径穿越漏洞**
- `zipfile` 默认 `ZIP_STORED` 不压缩，必须显式 `ZIP_DEFLATED`
- 备份要先 `df` 查空间，压缩包写到一半磁盘满会产生损坏文件
- `.tar.gz` 用 `"w:gz"` 而不是 `"w" + gzip` 两次包装

## 3. pandas ★★（按需） —— 数据聚合、筛选、统计

日志/指标的聚合统计。**只学这几个操作就够日常用**。

```
pd.read_csv(path, sep=, encoding=, parse_dates=, dtype=)
pd.DataFrame(dict) / DataFrame(list_of_dict)
df.head(n) / df.tail(n) / df.shape / df.columns / df.dtypes
df["col"]                     取列
df[df["status"] == 500]       过滤
df.loc[行条件, "列名"]         按标签
df.iloc[0:5, 0:2]             按位置
df.sort_values("col", ascending=False)
df.groupby("host")["cpu"].mean()
df.groupby(["host","status"]).agg({"size": ["sum","count"]})
df.merge(other, on="id", how="left")
df.fillna(0) / df.dropna(subset=["col"])
df["new"] = df["a"] + df["b"]
df.to_csv(path, index=False, encoding="utf-8-sig")
pd.to_datetime(df["ts"])
```

```Python
import pandas as pd

# ==== 1. 读访问日志（假设已解析成 CSV）====
df = pd.read_csv("access.csv", encoding="utf-8",
                 parse_dates=["time"])       # 自动转 datetime

# ==== 2. 概览 ====
print(df.shape, df.columns.tolist())
print(df.head(3))

# ==== 3. 过滤 ====
errors = df[df["status"] >= 500]

# ==== 4. 聚合：每个 host 的错误数、平均响应时间 ====
summary = (
    df.groupby("host")
      .agg(requests=("path", "count"),
           errors=("status", lambda s: (s >= 500).sum()),
           avg_size=("size", "mean"))
      .reset_index()
)
summary["error_rate"] = summary["errors"] / summary["requests"]

# ==== 5. 排序取 Top ====
top = summary.sort_values("errors", ascending=False).head(10)
print(top)

# ==== 6. 按小时统计请求量 ====
df["hour"] = df["time"].dt.floor("h")
per_hour = df.groupby("hour").size()

# ==== 7. 导出（utf-8-sig 让 Excel 不乱码）====
summary.to_csv("report.csv", index=False, encoding="utf-8-sig")
```

⚠️ 坑：

- `df["col"] == x` 返回的是布尔 Series，**不能直接 `if`**，要用 `.any()` / `.all()`
- 链式赋值（`df[df.a > 1]["b"] = 0`）不生效，用 `.loc`
- `to_csv` 加 `index=False`，否则多一列无名索引
- Excel 打开中文 CSV 要 `utf-8-sig`
- pandas 体积大（几十 MB），只在小机器/临时脚本里要慎重

## 4. cryptography ★★ —— 敏感配置加解密

比 `hashlib` 更进一步：可逆加解密、签名、证书。

```
from cryptography.fernet import Fernet

key = Fernet.generate_key()          生成密钥（32 字节 base64）
f = Fernet(key)
token = f.encrypt(b"secret")         加密
f.decrypt(token)                     解密（密钥错/被改会抛 InvalidToken）
```

```Python
import os
from cryptography.fernet import Fernet, InvalidToken

# ==== 密钥从环境变量/密钥管理拿，不要写代码里 ====
KEY = os.environ["CONF_KEY"].encode()      # Fernet.generate_key() 的结果
fernet = Fernet(KEY)

# ==== 加密写入配置文件 ====
plaintext = b"db_password=xxx"
Path("/etc/app/secret.enc").write_bytes(fernet.encrypt(plaintext))

# ==== 解密读取 ====
try:
    data = fernet.decrypt(Path("/etc/app/secret.enc").read_bytes())
except InvalidToken:
    raise SystemExit("密钥不对或文件被篡改")
```

```Python
# ==== 更细粒度：AES-GCM（要自己管 nonce）====
from cryptography.hazmat.primitives.ciphers.aead import AESGCM

key = AESGCM.generate_key(bit_length=256)
aes = AESGCM(key)
nonce = os.urandom(12)                       # 每次加密都必须换
ct = aes.encrypt(nonce, b"secret", None)
pt = aes.decrypt(nonce, ct, None)
```

⚠️ 坑：

- `Fernet.generate_key()` 每次调用都生成新密钥，**密钥丢了数据就永远解不开**
- 不要拿 `base64` 当加密用
- 加密密钥不能和密文放同一个仓库
- 面向前场：加密解决「存哪安全」，`hashlib` 只解决「有没有被改」

## 5. openpyxl ★（按需） —— 读写 xlsx，导出报表

```
from openpyxl import Workbook, load_workbook

wb = Workbook()                      新建
wb.active                            活动表
wb.create_sheet("名字")               加表
wb.sheetnames                        表名列表
ws = wb["名字"] / wb.active

load_workbook(path, read_only=True, data_only=True)
    read_only=True   大文件只读，省内存（但不能随机访问）
    data_only=True   读公式的计算结果而不是公式本身

ws.cell(row=1, column=1, value="表头")
ws["A1"] = "值"                       单元格直接赋值
ws.append([...])                     追加一行
ws.max_row / ws.max_column
ws.iter_rows(min_row=2, values_only=True)   逐行迭代（推荐）
ws.freeze_panes = "A2"               冻结首行
wb.save(path)
```

```Python
from openpyxl import Workbook, load_workbook
from openpyxl.styles import Font, PatternFill

# ==== 1. 写报表 ====
wb = Workbook()
ws = wb.active
ws.title = "巡检结果"

headers = ["节点", "状态", "CPU%", "内存%"]
ws.append(headers)
for cell in ws[1]:
    cell.font = Font(bold=True)
    cell.fill = PatternFill("solid", fgColor="DDDDDD")

rows = [
    ("k8s-01", "Ready",    42.1, 61.0),
    ("k8s-02", "NotReady", 97.5, 92.3),
]
for r in rows:
    ws.append(r)

ws.freeze_panes = "A2"
wb.save("/tmp/report.xlsx")

# ==== 2. 读大文件：只读模式 + 逐行迭代 ====
wb = load_workbook("/tmp/report.xlsx", read_only=True, data_only=True)
ws = wb["巡检结果"]
for row in ws.iter_rows(min_row=2, values_only=True):
    node, status, cpu, mem = row
    if status != "Ready":
        print(f"[异常] {node} {status} cpu={cpu} mem={mem}")
wb.close()          # 只读模式必须 close，否则文件句柄不释放
```

⚠️ 坑：

- 大文件一定要 `read_only=True`，否则整个工作簿读进内存
- `data_only=True` 只有在**文件被 Excel 保存过**后才拿得到公式结果，否则是 `None`
- 写 `.xlsx` 不能用 `csv` 模块，两者格式完全不同
- 单元格里写时间要用 `datetime` 对象，写字符串 Excel 会当文本
