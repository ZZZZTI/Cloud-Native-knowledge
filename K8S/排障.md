>

### 基础配置

```Shell
# 基础格式
kubectl [command] [TYPE] [NAME] [flags]
# -n <ns>                指定命名空间
# -A, --all-namespaces   所有命名空间
# -o wide                宽格式输出（多显示 IP、Node 等）
# -o yaml / -o json      以 YAML/JSON 输出完整对象定义
# -o jsonpath='{...}'    按 JSONPath 提取字段
# -l <selector>          按标签筛选，如 -l app=nginx,tier=frontend
# --show-labels          显示标签列
# -f <file/dir/url>      指定清单文件、目录或 URL
# --dry-run=client -o yaml  生成 YAML 但不提交（用于快速生成模板）
# -w, --watch            持续监听资源变化
# --sort-by=<field>      按指定字段排序
# -h, --help             查看命令帮助
# 类型简写：po=pod | svc=service | deploy=deployment | ns=namespace | no=node

# 集群配置与上下文
kubectl version                              # 查看版本
kubectl cluster-info                         # 查看集群信息
kubectl api-resources                        # 查看支持的资源类型
kubectl explain pod.spec                     # 查看资源字段说明
kubectl api-versions                         # 列出支持的 API 版本
kubectl config view                          # 查看合并后的 kubeconfig 配置
kubectl config get-contexts                  # 列出所有上下文（集群+用户+命名空间组合）
kubectl config current-context               # 显示当前使用的上下文
kubectl config use-context <context-name>    # 切换到指定上下文
kubectl config set-context --current --namespace=<ns>  # 切换当前上下文的默认命名空间
```

### 修改资源

```Shell
# 创建和更新
kubectl apply -f <file.yaml>                 # 声明式创建/更新（生产环境推荐）
kubectl apply -f ./dir/                      # 应用目录下所有 YAML
kubectl apply -f <url>                       # 从 URL 应用清单
kubectl create -f <file.yaml>                # 命令式创建（仅创建，不更新；已存在会报错）
kubectl create deployment nginx --image=nginx --replicas=3  # 快速创建 Deployment
kubectl run <pod> --image=nginx              # 快速运行一个 Pod（调试用）
kubectl expose deployment <deploy> --port=80 --target-port=8080 --type=NodePort  # 为 Deployment 创建 Service
kubectl create ns <ns>                       # 创建命名空间
kubectl create configmap <cm> --from-literal=key1=value1 --from-literal=key2=value2  # 从字面量创建 ConfigMap
kubectl create configmap <cm> --from-file=./config-dir/  # 从目录创建 ConfigMap
kubectl create secret generic <secret> --from-literal=password=123456  # 创建通用 Secret
kubectl create job hello --image=busybox -- echo "Hello"  # 创建一次性 Job
kubectl create cronjob hello --image=busybox --schedule="*/1 * * * *" -- date  # 创建 CronJob
kubectl set image deployment/<deploy> <container>=<new-image>  # 更新 Deployment 的容器镜像
kubectl scale deployment <deploy> --replicas=5  # 扩缩容 Deployment
kubectl autoscale deployment <deploy> --min=2 --max=10 --cpu-percent=80  # 创建 HPA 自动扩缩
kubectl label pod <pod> env=prod             # 为资源添加/修改标签
kubectl annotate pod <pod> description="web" # 为资源添加/修改注解
kubectl edit deploy/<name> -n <ns>           # 在线编辑

# 删除
kubectl delete pod <pod>                     # 删除 Pod（若由 Deployment 管理，会被重建）
kubectl delete deployment <deploy>           # 删除 Deployment（连带其管理的 Pod）
kubectl delete -f <file.yaml>                # 按清单文件删除资源
kubectl delete pods -l app=nginx             # 按标签批量删除 Pod
kubectl delete all --all -n <ns>             # 删除命名空间下常用资源（慎用，不会删 PV/PVC 等）
kubectl delete pod <pod> --grace-period=0 --force   # 强制删除，慎用
```

### 其他控制

```Shell
# Deployment发布
kubectl rollout status deployment/<deploy>   # 查看发布进度
kubectl rollout history deployment/<deploy>  # 查看发布历史
kubectl rollout history deployment/<deploy> --revision=3  # 查看指定版本的详情
kubectl rollout undo deployment/<deploy>     # 回滚到上一版本
kubectl rollout undo deployment/<deploy> --to-revision=2  # 回滚到指定版本
kubectl rollout restart deployment/<deploy>  # 滚动重启（不改变镜像，常用于让 Pod 重新拉取配置）
kubectl rollout pause deployment/<deploy>    # 暂停发布
kubectl rollout resume deployment/<deploy>   # 恢复发布

# 调度控制
kubectl cordon <node>                        # 将节点标记为不可调度（新 Pod 不会调度上去）
kubectl uncordon <node>                      # 解除不可调度标记
kubectl drain <node> --ignore-daemonsets --delete-emptydir-data  # 驱逐节点上的 Pod（维护前使用）
kubectl taint node <node> key=value:NoSchedule   # 给节点添加污点
kubectl taint node <node> key=value:NoSchedule-  # 移除污点（末尾加 -）
kubectl label node <node> disktype=ssd       # 给节点打标签

# 权限/诊断
kubectl proxy                                # 本地代理 API Server
kubectl auth can-i --list                    # 查看当前身份在各资源上的权限
kubectl auth whoami                          # 显示当前认证身份（k8s ≥1.27）
kubectl cluster-info dump | less             # 导出集群诊断信息，分页查看
kubectl api-resources --verbs=list --namespaced -o name \
| xargs -n1 kubectl get --show-kind --ignore-not-found -n <ns>  # 列出该 ns 下所有可 list 的资源

# 网络与端口
kubectl expose deploy/<name> -n <ns> --port=80 --target-port=8080  # 为 Deployment 创建 Service
kubectl port-forward svc/<svc> 8080:80 -n <ns>     # 本地 8080 → Service 80
kubectl port-forward pod/<pod> 8080:80 -n <ns>     # 本地 8080 → Pod 80

# 节点维护
kubectl cordon <node>                    # 标记不可调度
kubectl drain <node> --ignore-daemonsets --delete-emptydir-data
kubectl uncordon <node>                  # 恢复调度
kubectl taint nodes <node> key=value:NoSchedule
kubectl top nodes                        # 节点资源使用
kubectl top pods -n <ns>                 # Pod 资源使用
```

### 排障查看

```Shell
# 查看（etcd 里的期望状态）
kubectl get nodes                            # 查看所有节点
kubectl get pods                             # 查看当前命名空间下的 Pod
kubectl get pods -A                          # 查看所有命名空间下的 Pod（-A = --all-namespaces）
kubectl get pods -n <ns> -o wide             # 查看指定命名空间的 Pod，宽格式显示 IP/Node 等
kubectl get deploy,svc,ing,pvc,cm,secret     # 一次查看多种资源类型
kubectl get endpointslices                   # 新版端点切片
kubectl get endpoints <svc>                  # 检查 Service 后端
kubectl get all                              # 查看当前命名空间下的常用资源（并非真正“全部”）
kubectl get pod <pod> -o yaml                # 以 YAML 格式导出 Pod 完整定义
kubectl get pod <pod> -o json                # 以 JSON 格式导出
kubectl get pods --show-labels               # 显示 Pod 的标签
kubectl get pods -l app=nginx                # 按标签筛选 Pod
kubectl get pods --sort-by='.status.containerStatuses[0].restartCount'  # 按重启次数排序
kubectl get events --sort-by='.lastTimestamp'  # 按时间倒序查看事件（排障必用）

# 描述与日志（实际状态）
kubectl describe pod <pod> -n <ns>           # 查看 Pod 详细状态、事件、容器信息：看 Events
kubectl describe node <node>                 # 查看节点资源分配、污点、Conditions
kubectl explain pod                          # 查看 Pod 资源的字段文档
kubectl explain pod.spec.containers          # 查看嵌套字段的文档说明
kubectl cluster-info dump                    # 导出集群诊断信息
kubectl logs <pod>                           # 查看 Pod 日志（单容器）
kubectl logs <pod> -c <container>            # 查看 Pod 中指定容器的日志
kubectl logs -f <pod>                        # 实时跟踪日志（follow）
kubectl logs <pod> --previous                # 查看上次崩溃容器的日志（CrashLoopBackOff 必用）
kubectl logs <pod> --tail=100                # 只看最后 100 行
crictl ps -a                                 # 查看所有容器
crictl logs <id>                             # 查看容器日志
kubectl top pods                             # 查看 Pod CPU/内存实时用量（需 metrics-server）
kubectl top pods -A --sort-by=cpu            # 所有命名空间 Pod 按 CPU 排序
kubectl top nodes                            # 查看节点资源用量

# 调试
kubectl exec -it <pod> -- /bin/bash          # 进入容器交互式 Shell
kubectl exec -it <pod> -c <container> -- sh  # 进入指定容器（无 bash 时用 sh）
kubectl port-forward pod/<pod> 8080:80       # 将本地 8080 转发到 Pod 的 80
kubectl port-forward svc/<svc> 8080:80       # 将本地 8080 转发到 Service 的 80
kubectl cp <pod>:/path/in/pod ./local-path   # 从 Pod 复制文件到本地
kubectl cp ./local-path <pod>:/path/in/pod   # 从本地复制文件到 Pod
kubectl debug <pod> -it --image=busybox      # 注入临时调试容器（不改变原 Pod 镜像）
kubectl debug node/<node> -it --image=busybox # 在节点上创建调试 Pod
kubectl auth can-i get pods --as=system:serviceaccount:<ns>:<sa>  # 检查某个 ServiceAccount 的权限

# 排障流程
# 1) 先看 Pod 状态和重启次数
kubectl get pods -o wide
# 2) 看 describe 中的 Events 和容器状态
kubectl describe pod <pod>
# 3) 看当前日志
kubectl logs <pod> --tail=200
# 4) 若 CrashLoopBackOff，看上次崩溃日志
kubectl logs <pod> --previous
# 5) 若怀疑资源不足，看用量
kubectl top pod <pod> 2>/dev/null || echo "metrics-server 未安装"
# 6) 若仍无法定位，注入调试容器
kubectl debug <pod> -it --image=busybox --target=<container>

1. 看现象：是 Pod 问题？Service 问题？Ingress 问题？节点问题？应用问题？
2. kubectl get：确认资源状态（Running/Pending/CrashLoopBackOff）
3. kubectl describe：看 Events，找关键错误信息
4. kubectl logs / --previous：看应用日志
5. kubectl exec：进容器验证配置、网络、文件
6. 查 Service/Endpoints：Endpoints 是否为空？selector 是否匹配？
7. 查 DNS：nslookup / dig 测试域名解析
8. 查 NetworkPolicy：是否被隔离？
9. 查存储：PVC 是否 Bound？PV 是否可用？
10. 查节点/控制面：kubelet、apiserver、etcd 是否正常？

# 错误码
码  含义	              典型场景
0	  正常退出	         Job 完成；前台进程退出也会 0 → CrashLoopBackOff
1	  应用报错退出	      配置错误、依赖连不上、参数不合法
126	命令不可执行	      入口脚本没有可执行权限
127	命令找不到	command  写了镜像里没有的二进制
137	128+9(SIGKILL)	  OOMKilled 或 kubectl delete 强杀
143	128+15(SIGTERM)	  优雅终止超时后被杀
139	SIGSEGV	          程序段错误（架构不匹配最常见）
```

### 排障操作

| 现象                         | 操作                                 | 常见原因                                                     |
| ---------------------------- | ------------------------------------ | ------------------------------------------------------------ |
| `ImagePullBackOff`           | `describe pod` 看 Events             | 镜像名/标签错、私有仓库缺 imagePullSecret、**网络不通/加速失效** |
| `CrashLoopBackOff`           | `logs --previous`                    | 应用配置错、依赖服务连不上、探针太激进、命令写错             |
| `Pending`                    | `describe pod` 看 Events             | 资源不足、亲和性/污点不满足、PVC Pending、超配额             |
| `OOMKilled`                  | `describe pod` 看 Last State         | `limits` 太小、内存泄漏、JVM 堆未设上限                      |
| `Running` 但 Service 不通    | `get endpoints`                      | selector 不匹配、Pod 未 Ready、targetPort 错                 |
| `CreateContainerConfigError` | `describe pod`                       | 引用的 ConfigMap/Secret 不存在或 key 缺失                    |
| `Evicted`                    | `describe node` / `get events`       | 节点磁盘/内存压力，`kubelet` 驱逐                            |
| 无法访问 Ingress             | `describe ingress` + Controller 日志 | 没装 Controller、ingressClassName 错、后端 Service 名错      |
